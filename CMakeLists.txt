cmake_minimum_required(VERSION 3.16)
project(phaser360_sof_offline LANGUAGES CXX)

option(PHASER_SOF_SANITIZERS "Instrument user-mode tests with ASan/UBSan" OFF)
add_library(phaser360_sof_parser STATIC src/sof/firmware_image.cpp)
add_library(phaser360_sof_boot STATIC src/sof/boot_sequence.cpp src/sof/ipc3_ready.cpp)
target_link_libraries(phaser360_sof_boot PUBLIC phaser360_sof_parser)
add_executable(phaser360_sof_boot_tests tests/sof_boot_tests.cpp)
target_link_libraries(phaser360_sof_boot_tests PRIVATE phaser360_sof_boot)
target_include_directories(phaser360_sof_parser PUBLIC src/sof)
target_compile_features(phaser360_sof_parser PUBLIC cxx_std_17)
add_executable(phaser360_sof_tests tests/sof_firmware_tests.cpp)
add_executable(phaser360_sof_inspect tools/sof_inspect.cpp)
add_executable(phaser360_m051_resource_tests tests/m051_resource_tests.cpp)
target_compile_features(phaser360_m051_resource_tests PRIVATE cxx_std_17)
target_link_libraries(phaser360_sof_tests PRIVATE phaser360_sof_parser)
target_link_libraries(phaser360_sof_inspect PRIVATE phaser360_sof_parser)

add_library(phaser360_hda_bdl STATIC src/sof/hda_bdl.cpp src/sof/hda_stream.cpp src/sof/glk_rom.cpp src/sof/ipc3_ready.cpp src/sof/ipc3_receive.cpp src/sof/ipc3_command.cpp)
add_executable(phaser360_glk_boot_tests tests/sof_glk_boot_tests.cpp m062/driver/cold_power.cpp m062/driver/ipc_interrupt.cpp m062/driver/glk_boot.cpp m062/driver/hda_transport.cpp m062/driver/boot_dma.cpp)
target_include_directories(phaser360_glk_boot_tests PRIVATE tests/fake_wdf)
target_link_libraries(phaser360_glk_boot_tests PRIVATE phaser360_hda_bdl)
add_executable(phaser360_command_tests tests/sof_command_tests.cpp)
target_link_libraries(phaser360_command_tests PRIVATE phaser360_hda_bdl)
add_executable(phaser360_ipc_tests tests/sof_ipc_tests.cpp)
target_link_libraries(phaser360_ipc_tests PRIVATE phaser360_hda_bdl)
add_executable(phaser360_rom_tests tests/sof_rom_tests.cpp)
target_link_libraries(phaser360_rom_tests PRIVATE phaser360_hda_bdl)
add_executable(phaser360_stream_tests tests/sof_stream_tests.cpp)
target_link_libraries(phaser360_stream_tests PRIVATE phaser360_hda_bdl)
target_compile_features(phaser360_hda_bdl PUBLIC cxx_std_17)
add_executable(phaser360_bdl_tests tests/sof_bdl_tests.cpp)
target_link_libraries(phaser360_bdl_tests PRIVATE phaser360_hda_bdl)
add_executable(phaser360_dma_owner_tests tests/sof_dma_owner_tests.cpp m062/driver/boot_dma.cpp)
target_include_directories(phaser360_dma_owner_tests PRIVATE tests/fake_wdf)
target_link_libraries(phaser360_dma_owner_tests PRIVATE phaser360_hda_bdl)

foreach(target phaser360_command_tests phaser360_ipc_tests phaser360_glk_boot_tests phaser360_rom_tests phaser360_stream_tests phaser360_hda_bdl phaser360_bdl_tests phaser360_dma_owner_tests phaser360_sof_boot phaser360_sof_boot_tests phaser360_sof_parser phaser360_sof_tests phaser360_sof_inspect phaser360_m051_resource_tests)
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4 /WX /permissive-)
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Werror)
        if(PHASER_SOF_SANITIZERS)
            target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-sanitize-recover=all -fno-omit-frame-pointer)
            target_link_options(${target} PRIVATE -fsanitize=address,undefined)
        endif()
    endif()
endforeach()

enable_testing()
add_test(NAME sof_command COMMAND phaser360_command_tests)
add_test(NAME sof_ipc COMMAND phaser360_ipc_tests)
add_test(NAME sof_glk_boot COMMAND phaser360_glk_boot_tests)
add_test(NAME sof_rom COMMAND phaser360_rom_tests)
add_test(NAME sof_stream COMMAND phaser360_stream_tests)
add_test(NAME sof_bdl COMMAND phaser360_bdl_tests)
add_test(NAME sof_dma_owner COMMAND phaser360_dma_owner_tests)
add_test(NAME sof_boot_protocol COMMAND phaser360_sof_boot_tests)
set_tests_properties(sof_boot_protocol PROPERTIES TIMEOUT 60)
add_test(NAME sof_parser_synthetic COMMAND phaser360_sof_tests)
add_test(NAME m051_resource_contract COMMAND phaser360_m051_resource_tests)
if(WIN32)
    add_executable(phaser360_m051_setupapi_abi_tests tests/m051_setupapi_abi_tests.cpp)
    target_compile_features(phaser360_m051_setupapi_abi_tests PRIVATE cxx_std_17)
    target_compile_options(phaser360_m051_setupapi_abi_tests PRIVATE /W4 /WX)
    add_test(NAME m051_setupapi_abi COMMAND phaser360_m051_setupapi_abi_tests)
endif()
set_tests_properties(sof_parser_synthetic PROPERTIES TIMEOUT 60)
